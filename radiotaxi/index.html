function doGet(e) {
  var action = e.parameter.action;
  var idLinea = e.parameter.idLinea || "radiotaximetro";
  
  if (action === "getLibres") {
    var libres = obtenerConductoresLibres(idLinea);
    return ContentService.createTextOutput(JSON.stringify(libres))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "OK", message: "API activa" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.action === "asignar") {
      var resultado = cambiarEstadoConductor(data.idLinea, data.conductorNombre, "OCUPADO");
      return ContentService.createTextOutput(JSON.stringify(resultado))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ exito: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Obtiene 煤nicamente los m贸viles con estado LIBRE
function obtenerConductoresLibres(idLinea) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Configuracion");
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var libres = [];

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === idLinea) {
      // Bloques arrancan en F (col 5): b1 desc, b1 link, b1 estado...
      for (var col = 5; col < headers.length; col += 3) {
        var nombre = data[i][col];
        var telefono = data[i][col+1];
        var estado = data[i][col+2];

        if (nombre && estado === "LIBRE") {
          libres.push({
            nombre: nombre,
            telefono: telefono,
            bloque: headers[col].substring(0, 2)
          });
        }
      }
      break;
    }
  }
  return libres;
}

// Modifica el estado en la celda correspondiente
function cambiarEstadoConductor(idLinea, conductorNombre, nuevoEstado) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Configuracion");
  var data = sheet.getDataRange().getValues();
  var headers = data[0];

  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === idLinea) {
      for (var col = 5; col < headers.length; col += 3) {
        if (data[i][col] === conductorNombre) {
          // La columna de estado es col + 2
          sheet.getRange(i + 1, col + 3).setValue(nuevoEstado);
          return { exito: true, conductor: conductorNombre, estado: nuevoEstado };
        }
      }
    }
  }
  return { exito: false, mensaje: "Conductor no encontrado" };
}
